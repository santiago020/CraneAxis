───────────────────────────────────────────────────────────────────────────── class SerialWorker(QObject): data_received = pyqtSignal(dict) status_changed = pyqtSignal(str, str) # mensaje, color def __init__(self): super().__init__() self.serial_port = None self.running = False self.connected = False self._lock = threading.Lock() def connect(self, port, baud=BAUD_RATE): try: self.serial_port = serial.Serial(port, baud, timeout=0.1) self.connected = True self.running = True self.status_changed.emit(f"Conectado a {port} @ {baud} baud", ACCENT_GREEN) thread = threading.Thread(target=self._read_loop, daemon=True) thread.start() return True except Exception as e: self.status_changed.emit(f"Error al conectar: {e}", ACCENT_RED) return False def disconnect(self): self.running = False self.connected = False if self.serial_port and self.serial_port.is_open: self.serial_port.close() self.status_changed.emit("Desconectado", ACCENT_ORANGE) def send_command(self, cmd: str): if self.connected and self.serial_port and self.serial_port.is_open: try: with self._lock: self.serial_port.write((cmd + "\n").encode()) except Exception as e: self.status_changed.emit(f"Error al enviar: {e}", ACCENT_RED) def _read_loop(self): """ Protocolo esperado desde ESP32 (CSV por linea): rk,yk,ek,uk,h1,h2 Ejemplo: 5.00,4.82,0.18,127.50,4.82,3.10 """ buffer = "" while self.running: try: if self.serial_port.in_waiting > 0: raw = self.serial_port.read(self.serial_port.in_waiting) buffer += raw.decode('utf-8', errors='ignore') lines = buffer.split('\n') buffer = lines[-1] for line in lines[:-1]: line = line.strip() if line: parsed = self._parse_line(line) if parsed: self.data_received.emit(parsed) else: time.sleep(0.01) except Exception: time.sleep(0.05) def _parse_line(self, line: str): try: parts = [float(x) for x in line.split(',')] if len(parts) >= 6: return { 'rk': parts[0], 'yk': parts[1], 'ek': parts[2], 'uk': parts[3], 'h1': parts[4], 'h2': parts[5], 'ts': time.time() } except Exception: pass return None # ───────────────────────────────────────────────────────────────────────────── # SIMULADOR (para pruebas sin hardware) # ───────────────────────────────────────────────────────────────────────────── class PlantSimulator(QObject): data_ready = pyqtSignal(dict) def __init__(self): super().__init__() self.running = False self.rk = 4.0 self.yk = 0.0 self.t = 0.0 self.signal_type = "square" self.period = 20.0 def start(self): self.running = True thread = threading.Thread(target=self._sim_loop, daemon=True) thread.start() def stop(self): self.running = False def set_setpoint(self, val): self.rk = val def set_signal(self, sig): self.signal_type = sig def _sim_loop(self): dt = 0.1 tau = 3.0 Kp = 0.8 ek_prev = 0.0 uk = 0.0 while self.running: self.t += dt rk_now = self._compute_reference(self.t) ek = rk_now - self.yk # Controlador P simple para simulacion uk = Kp * ek uk = max(0, min(255, uk * 30)) # Planta de primer orden self.yk += dt * ((-self.yk + uk / 30.0) / tau) self.yk = max(0, min(8.0, self.yk)) self.data_ready.emit({ 'rk': round(rk_now, 3), 'yk': round(self.yk, 3), 'ek': round(ek, 3), 'uk': round(uk, 3), 'h1': round(self.yk, 3), 'h2': round(self.yk * 0.6, 3), 'ts': time.time() }) time.sleep(dt) def _compute_reference(self, t): base = self.rk if self.signal_type == "square": cycle = (t % self.period) / self.period offset = 1.5 if cycle < 0.5 else -1.5 elif self.signal_type == "ramp": offset = ((t % self.period) / self.period) * 3.0 - 1.5 else: # sinusoidal offset = 1.5 * np.sin(2 * np.pi * t / self.period) return max(TANK_MIN_CM, min(TANK_MAX_CM, base + offset)) #
